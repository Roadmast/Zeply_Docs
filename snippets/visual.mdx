export const Visual = ({ kind = "screenshot", note, src, alt, caption }) => {
  if (src) {
    return (
      <Frame caption={caption}>
        <img src={src} alt={alt || note} />
      </Frame>
    );
  }
  const label = kind === "gif" ? "GIF" : "Screenshot";
  return (
    <div
      style={{
        border: "2px dashed rgba(128,128,128,0.5)",
        borderRadius: "12px",
        background: "rgba(128,128,128,0.08)",
        minHeight: "220px",
        margin: "16px 0",
        padding: "24px",
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        justifyContent: "center",
        textAlign: "center",
        gap: "6px",
      }}
    >
      <div style={{ fontSize: "12px", fontWeight: 600, letterSpacing: "0.08em", textTransform: "uppercase", opacity: 0.7 }}>
        {label} placeholder
      </div>
      <div style={{ fontSize: "14px", opacity: 0.85 }}>{note}</div>
    </div>
  );
};
