import { Visual } from "/snippets/visual.mdx";

WhatsApp lets a business start a conversation only with an approved template. You'll pick a contact and send one.

<Steps>
  <Step title="Open Inbox">
    Click **Inbox** in the sidebar.

    <Visual note="Inbox with the template button beside the search box highlighted" />
  </Step>
  <Step title="Click the template button">
    Click the template button beside the search box. **Start New Conversation** opens.

    <Visual note="Start New Conversation dialog" />
  </Step>
  <Step title="Choose a contact">
    Pick the person to message.

    <Visual note="Contact picker with one contact selected" />
  </Step>
  <Step title="Choose a template">
    Pick an approved WhatsApp template and fill in any variables.

    <Visual note="Template picker with a preview" />
  </Step>
  <Step title="Send it">
    Click send. The conversation appears in **Inbox**, and the customer's reply continues it.

    <Visual note="New conversation in the list" />
  </Step>
</Steps>

<Note>
No templates to choose from? Open **Message Templates** and [sync them from Meta](/guides/templates/sync).
</Note>
