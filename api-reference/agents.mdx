---
title: "Agents API"
api: "POST /api/v1/agents"
description: "Create and manage conversational voice agents"
---

### Headers

<ParamField header="Authorization" type="string" required>
  Bearer token: `Bearer <token>`
</ParamField>

### Body

<ParamField body="name" type="string" required>
  Descriptive name of the agent.
</ParamField>

<ParamField body="system_prompt" type="string" required>
  System instruction governing the agent's behavior and personality.
</ParamField>

<ParamField body="voice_id" type="string" default="cartesia/sonic-english">
  Voice model identifier.
</ParamField>

<ParamField body="temperature" type="number" default="0.3">
  LLM sampling temperature between `0.0` and `1.0`.
</ParamField>

### Response

<ResponseField name="id" type="string">
  Unique agent identifier.
</ResponseField>

<ResponseField name="name" type="string">
  Agent name.
</ResponseField>

<ResponseField name="status" type="string">
  Status code of the agent.
</ResponseField>

<ResponseExample>

{
  "id": "ag_7a1b89ef",
  "name": "Support Concierge",
  "voice_id": "cartesia/sonic-english",

</ResponseExample>
